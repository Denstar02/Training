Подключение базы. Нужно один раз на всё приложение: дальше людям хватит почты и пароля.

Как завести проект за пять минут
  • Зайди на supabase.com, Sign up, New project. План Free, пароль базы придумай любой.
  • Settings → API: скопируй Project URL и ключ anon public в поля выше.
  • Authentication → Providers → Email: выключи Confirm email, чтобы входить сразу.
  • SQL Editor → New query: вставь код ниже и нажми Run.
create table if not exists public.progress (
  user_id uuid primary key references auth.users on delete cascade,
  data jsonb not null,
  updated_at timestamptz not null default now()
);
alter table public.progress enable row level security;
create policy "read own" on public.progress
  for select using (auth.uid() = user_id);
create policy "insert own" on public.progress
  for insert with check (auth.uid() = user_id);
create policy "update own" on public.progress
  for update using (auth.uid() = user_id) with check (auth.uid() = user_id);

  title text not null check (char_length(title) between 1 and 40),
  note text check (char_length(note) <= 140),

  for select to authenticated using (true);
create policy "awards give" on public.awards
  for insert to authenticated with check (
    given_by = auth.uid()
    and exists (select 1 from public.badge_admins a where a.user_id = auth.uid()));
create policy "awards take" on public.awards
  for delete to authenticated using (
    exists (select 1 from public.badge_admins a where a.user_id = auth.uid()));

insert into public.badge_admins (user_id)
  select id from auth.users where lower(email) = lower('почта@админа')
  on conflict do nothing;

Чтобы друзей можно было добавлять по почте, — запрос для поиска друзей по почте. Он находит только по точному адресу и не показывает чужие почты:

create or replace function public.find_friend(q text)
returns table (user_id uuid, code text, nick text, photo text, stats jsonb)
language sql stable security definer set search_path = public as $$
  select p.user_id, p.code, p.nick, p.photo, p.stats
  from public.profiles p
  join auth.users u on u.id = p.user_id
  where auth.uid() is not null
    and lower(u.email) = lower(trim(q))
  limit 1;
$$;
revoke all on function public.find_friend(text) from public, anon;
grant execute on function public.find_friend(text) to authenticated;

Для открытого доступа — запрос, после которого профиль видят только взаимные друзья, дружба идёт через заявки, а ещё работают поиск, удаление аккаунта и скрытие ника или фото админом. Выполни его после остальных; повторный запуск ничего не сломает:

-- Открытый доступ: профиль видят только взаимные друзья, заявки в друзья,
-- поиск по коду, почте и нику, удаление аккаунта, скрытие ника и фото админом.
-- Запрос можно выполнять повторно.

-- 1. Профиль видят только сам владелец и взаимные друзья
drop policy if exists "profiles read" on public.profiles;
drop policy if exists "profiles read friends" on public.profiles;
create policy "profiles read friends" on public.profiles
  for select to authenticated using (
    user_id = auth.uid()
    or (exists (select 1 from public.friends f where f.user_id = auth.uid() and f.friend_id = profiles.user_id)
        and exists (select 1 from public.friends f where f.user_id = profiles.user_id and f.friend_id = auth.uid())));

-- 2. Заявку можно отклонить, дружбу — убрать с обеих сторон; себя в друзья не добавить
drop policy if exists "friends drop" on public.friends;
create policy "friends drop" on public.friends
  for delete to authenticated using (auth.uid() = user_id or auth.uid() = friend_id);
drop policy if exists "friends add" on public.friends;
create policy "friends add" on public.friends
  for insert to authenticated with check (auth.uid() = user_id and friend_id <> auth.uid());

-- 3. Значки видят владелец, тот, кто выдал, и взаимные друзья
do $$ begin
  if to_regclass('public.awards') is not null then
    execute 'drop policy if exists "awards read" on public.awards';
    execute $p$create policy "awards read" on public.awards
      for select to authenticated using (
        user_id = auth.uid() or given_by = auth.uid()
        or (exists (select 1 from public.friends f where f.user_id = auth.uid() and f.friend_id = awards.user_id)
            and exists (select 1 from public.friends f where f.user_id = awards.user_id and f.friend_id = auth.uid())))$p$;
  end if;
end $$;

-- 4. Поиск и заявки: наружу только ник, фото, уровень и код
create or replace function public.find_people(q text)
returns table (user_id uuid, code text, nick text, photo text, lv int)
language sql stable security definer set search_path = public as $$
  select p.user_id, p.code, p.nick, p.photo,
         case when p.stats->'lv'->>'n' ~ '^[0-9]{1,4}$' then (p.stats->'lv'->>'n')::int else 0 end
  from public.profiles p
  join auth.users u on u.id = p.user_id
  where auth.uid() is not null and p.user_id <> auth.uid()
    and char_length(trim(q)) between 2 and 80
    and (p.code = upper(trim(q))
         or lower(u.email) = lower(trim(q))
         or lower(trim(p.nick)) = lower(trim(q)))
  order by p.updated_at desc
  limit 6;
$$;

create or replace function public.my_people()
returns table (user_id uuid, code text, nick text, photo text, lv int)
language sql stable security definer set search_path = public as $$
  select p.user_id, p.code, p.nick, p.photo,
         case when p.stats->'lv'->>'n' ~ '^[0-9]{1,4}$' then (p.stats->'lv'->>'n')::int else 0 end
  from public.profiles p
  where auth.uid() is not null and p.user_id <> auth.uid()
    and exists (select 1 from public.friends f
                where (f.user_id = auth.uid() and f.friend_id = p.user_id)
                   or (f.user_id = p.user_id and f.friend_id = auth.uid()));
$$;

-- 5. Удалить свой аккаунт со всеми данными
create or replace function public.delete_me()
returns void language plpgsql security definer set search_path = public as $$
declare me uuid := auth.uid();
begin
  if me is null then raise exception 'not signed in'; end if;
  if to_regclass('public.awards') is not null then
    execute 'delete from public.awards where given_by = $1' using me;
  end if;
  delete from auth.users where id = me;
end;
$$;

-- 6. Модерация: админ скрывает ник или фото; тот же ник или то же фото снова не появятся
create table if not exists public.badge_admins (
  user_id uuid primary key references auth.users on delete cascade
);
alter table public.badge_admins enable row level security;
create table if not exists public.profile_bans (
  user_id uuid primary key references auth.users on delete cascade,
  nick text,
  photo_md5 text,
  by_admin uuid references auth.users on delete set null,
  created_at timestamptz not null default now()
);
alter table public.profile_bans enable row level security;
grant select on public.profile_bans to authenticated;
drop policy if exists "bans read" on public.profile_bans;
create policy "bans read" on public.profile_bans
  for select to authenticated using (
    user_id = auth.uid()
    or exists (select 1 from public.badge_admins a where a.user_id = auth.uid()));

create or replace function public.profiles_apply_bans() returns trigger
language plpgsql security definer set search_path = public as $$
declare b public.profile_bans%rowtype;
begin
  select * into b from public.profile_bans where user_id = new.user_id;
  if found then
    if b.nick is not null and lower(trim(coalesce(new.nick, ''))) = lower(trim(b.nick)) then new.nick := null; end if;
    if b.photo_md5 is not null and new.photo is not null and md5(new.photo) = b.photo_md5 then new.photo := null; end if;
  end if;
  return new;
end;
$$;
drop trigger if exists profiles_apply_bans on public.profiles;
create trigger profiles_apply_bans before insert or update on public.profiles
  for each row execute function public.profiles_apply_bans();

create or replace function public.admin_hide(target uuid, what text)
returns void language plpgsql security definer set search_path = public as $$
declare p public.profiles%rowtype;
begin
  if not exists (select 1 from public.badge_admins a where a.user_id = auth.uid()) then
    raise exception 'only admin';
  end if;
  select * into p from public.profiles where user_id = target;
  if not found then raise exception 'no profile'; end if;
  insert into public.profile_bans (user_id, by_admin) values (target, auth.uid())
    on conflict (user_id) do update set by_admin = excluded.by_admin, created_at = now();
  if what = 'nick' then
    update public.profile_bans set nick = p.nick where user_id = target;
    update public.profiles set nick = null where user_id = target;
  elsif what = 'photo' then
    update public.profile_bans set photo_md5 = md5(p.photo) where user_id = target;
    update public.profiles set photo = null where user_id = target;
  else
    raise exception 'what: nick or photo';
  end if;
end;
$$;

Правила, без которых программа не работает

  • Никогда не подтягивайся до отказа. Заканчивай подход, когда можешь сделать ещё 1–2 повторения.
  • Перед первым подходом разогрей плечи: круги руками, 10–15 секунд просто повиси на перекладине.
  • Техника: из полного виса сначала опусти лопатки, потом тянись грудью к перекладине, подбородок выше неё. Без раскачки.
  • Удержание вверху считается как одно подтягивание: подтянулся, задержался, медленно опустился за 2–3 секунды.
  • Если болят локти или плечи — пропусти тренировку. Тренировки идут подряд, ничего не сгорает.
  • Подтягивания нагружают спину и бицепс. Для роста мышц нужны достаточно еды и белка и сон 7–8 часов, а для баланса делай отжимания из соседней вкладки.

Правила для отжиманий

  • Как и на турнике, не доходи до отказа: заканчивай подход, когда в запасе ещё 2–3 повторения.
  • Техника: ладони под плечами, локти идут назад под углом 45 градусов, тело прямое, таз не проваливается и не задирается.
  • Если с пола тяжело, делай с упором в стол или подоконник: чем выше опора, тем легче. Максимум тогда измеряй в том же варианте.
  • Тренировки идут подряд, а не по дням недели. Пропустил день — ничего не сгорает, просто продолжай со следующей.
  • После медленных отжиманий мышцы могут побаливать: тогда сделай паузу в день. Лёгкую тренировку можно делать хоть каждый день.
  • Отжимания и подтягивания не мешают друг другу: работают противоположные мышцы.

Правила для приседаний

  • Не доходи до отказа: заканчивай подход, когда в запасе ещё 2–3 повторения.
  • Техника: стопы на ширине плеч и слегка развёрнуты, пятки не отрываются, колени идут в стороны по носкам, спина прямая, взгляд вперёд.
  • Глубина: таз опускается хотя бы до уровня колен. Если не хватает подвижности, приседай до комфортной глубины и постепенно увеличивай её.
  • Если колени болят, уменьши глубину и число повторений, а при боли, которая не проходит, покажись врачу.
  • Приседания хорошо сочетаются с турником и отжиманиями: ноги отдыхают, пока работает верх тела.
  • Тренировки идут подряд, а не по дням недели. Пропустил день — ничего не сгорает, просто продолжай со следующей.

Правила для гантелей

  • Максимум считается в жиме гантелей лёжа: сколько чистых повторений выходит с твоим весом. От него строятся все подходы.
  • Не доходи до отказа: заканчивай подход, когда в запасе ещё 2–3 повторения.
  • Спина всегда прямая, движение без рывков, опускание медленнее подъёма.
  • Когда план начнёт даваться легко, бери гантели потяжелее и заново измерь максимум в день теста.
  • В выпадах и тягах считается подход на каждую сторону, чередуй ноги и руки.
  • Если гантель одна, делай подходы поочерёдно на каждую руку с тем же числом повторений.

Запланировано, нажми после подхода Выполнено Меньше плана Секунды удержания

Правила растяжки

  • Всегда начинай с разминки. На холодные мышцы тянуться бесполезно и травмоопасно.
  • Тяни до ощущения натяжения, а не боли. По десятибалльной шкале это 5–6 из 10: терпимо, можно спокойно дышать и говорить.
  • Не пружинь и не качайся рывками. Опустился, расслабился, дышишь ровно, держишь 30–60 секунд.
  • Выдох помогает: на выдохе мышца отпускает, и получается опуститься чуть ниже.
  • Под колени, таз и руки подкладывай подушки, пледы и книги. Опора — не поблажка, а способ расслабиться и дать мышце удлиниться.
  • Резкая или стреляющая боль, особенно в задней поверхности бедра у таза или в паху, — сразу прекращай. Такие надрывы заживают месяцами.
  • Месяц ежедневных занятий заметно прибавляет подвижность, но полный шпагат за 30 дней садится далеко не у всех. Ориентируйся на сантиметры в замерах, а не на срок.

Редактор программы

Как это работает
Запланировано, нажми после подхода Выполнено Меньше плана Секунды удержания
  • Программа состоит из дней. Каждый день — это список упражнений с числом подходов и повторений или секунд.
  • Дни идут по кругу. Два дня и двенадцать занятий в цикле — значит шесть повторов каждого дня.
  • Отдых задаётся в минутах для каждого упражнения. Ноль означает, что таймер не запускается: удобно для растяжки и круговых подходов.
  • Отметки ставятся так же, как в остальных программах: нажал на кружок — подход засчитан, можно поправить число в редакторе под ним.
  • Изменения сохраняются сразу и попадают в синхронизацию вместе с остальным прогрессом.
  • Программ может быть несколько: переключай их кнопками сверху. Прогресс у каждой свой.
  • Программой можно поделиться: в редакторе нажми «Поделиться программой» и отправь ссылку. Получатель откроет её, и программа добавится к нему, без твоих отметок. Если ссылка не открывается в нужном месте, в «Создать» есть «Добавить по коду» — туда можно вставить ссылку целиком.
